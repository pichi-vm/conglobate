mv /var/lib/random-seed /var/lib/systemd/random-seed &>/dev/null
mv /var/lib/backlight /var/lib/systemd/backlight &>/dev/null
if [ -L /var/lib/systemd/timesync ]; then
    rm /var/lib/systemd/timesync
    mv /var/lib/private/systemd/timesync /var/lib/systemd/timesync
fi
if [ -f /var/lib/systemd/clock ]; then
    mkdir -p /var/lib/systemd/timesync
    mv /var/lib/systemd/clock /var/lib/systemd/timesync/.

systemd-hwdb update &>/dev/null

    grep -v 'Failed to open /dev/urandom' || :

posttrans scriptlet (using /bin/sh):
# Restart some services.
# Others are either oneshot services, or sockets, and restarting them causes issues (#1378974)

 
if [ $1 -ge 2 ] && [ -x "/usr/lib/systemd/systemd-update-helper" ]; then 
    # Package upgrade, not install 
    /usr/lib/systemd/systemd-update-helper mark-restart-system-units systemd-udevd.service systemd-timesyncd.service systemd-homed.service systemd-oomd.service systemd-portabled.service || : 
fi
